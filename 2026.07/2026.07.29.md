### [[PCCP 모의고사 #1] 1번 - 외톨이 알파벳](https://school.programmers.co.kr/learn/courses/20847/lessons/255900)

```cpp
#include <string>
#include <vector>

using namespace std;

string solution(string input_string) {
    string answer = "";
    vector<int> last(26, -1);
    
    for(int i=0; i<input_string.length(); i++){
        char c = input_string[i];
        int index = c - 'a';
        if(last[index] == -1) last[index] = i;
        else last[index] = i - last[index] == 1 ? i : -2;
    }
    for(int i=0; i<last.size(); i++)
        if(last[i] == -2) answer += 'a' + i;
    return answer.length() == 0 ? "N" : answer;
}
```

* **문자열**, **구현** 문제
* `last`에 알파벳이 마지막으로 등장한 위치를 저장
* 현재 알파벳의 위치와 마지막 위치의 거리가 **`2` 이상이면 외톨이**

### [[PCCP 모의고사 #1] 3번 - 유전법칙](https://school.programmers.co.kr/learn/courses/20847/lessons/255902)

```cpp
#include <string>
#include <vector>

using namespace std;

vector<string> solution(vector<vector<int>> queries) {
    vector<string> answer;
    vector<vector<int>> tree = 
    {
        {1, 0, 0, 2},  // Rr
        {1, 1, 1, 1},  // RR
        {2, 2, 2, 2}   // rr
    };
    vector<string> gene = {"Rr", "RR", "rr"};
    
    for(int i=0; i<queries.size(); i++){
        vector<int> mod;
        int num = queries[i][1] - 1;
        for(int j=0; j<queries[i][0]; j++){
            mod.push_back(num % 4);
            num /= 4;
        }
        
        int curr = 0;
        for(int j=mod.size()-2; j>=0; j--) curr = tree[curr][mod[j]];
        answer.push_back(gene[curr]);
    }

    return answer;
}
```

* **진법 변환?**, **트리** 문제
* **자식 노드의 수**가 반드시 `4`개이므로 **나머지가 부모의 몇 번째 자식인지를 의미**

### [[PCCP 모의고사 #1] 4번 - 운영체제](https://school.programmers.co.kr/learn/courses/20847/lessons/255903)

```cpp
#include <string>
#include <vector>
#include <queue>
#include <algorithm>

using namespace std;

struct Data{
    int priority;
    long long startTime;
    long long execTime;
    
    Data(const vector<int>& A){
        priority = A[0];
        startTime = A[1];
        execTime = A[2];
    }
    
    bool operator<(const Data& d) const{
        if(priority != d.priority) return priority > d.priority;
        else return startTime > d.startTime;
    }
};

bool pushProcess(priority_queue<Data>& pq, const vector<vector<int>>& program, int& cur, long long& time){
    while(cur < program.size() && program[cur][1] <= time){
        pq.push(Data(program[cur]));
        cur++;
    }
    
    if(cur < program.size() && pq.empty()) time = program[cur][1];
    
    while(cur < program.size() && program[cur][1] <= time){
        pq.push(Data(program[cur]));
        cur++;
    }
    
    return !pq.empty();
}

vector<long long> solution(vector<vector<int>> program) {
    vector<long long> answer(11, 0);
    long long time = 0;
    int cur = 0;
    sort(program.begin(), program.end(), [](const vector<int>& A, const vector<int>& B){
        if(A[1] != B[1]) return A[1] < B[1];
        return A[0] < B[0];
    });
    
    priority_queue<Data> pq;
    
    while(pushProcess(pq, program, cur, time)){
        Data data = pq.top(); pq.pop();
        answer[data.priority] += (time - data.startTime);
        time += data.execTime;
    }
    answer[0] = time;
    return answer;
}
```

* **우선순위 큐**, **시뮬레이션** 유형의 문제
* 문제의 스케줄링 방법은 **비선점**이므로 실행 시간을 전부 더해줌
* 만약 대기 중인 프로세스가 없으면 **다음 프로세스 실행까지 시간을 점프**

### [[PCCP 모의고사 #2] 1번 - 실습용 로봇](https://school.programmers.co.kr/learn/courses/20848/lessons/255904)

```cpp
#include <string>
#include <vector>
using namespace std;

vector<int> solution(string command) {
    vector<int> answer(2, 0);
    int dir = 0;
    vector<vector<int>> move(4, vector<int>(2, 0));
    move[0] = {0, 1}; move[1] = {1, 0}; move[2] = {0, -1}; move[3] = {-1, 0};
    
    for(int i=0; i<command.length(); i++){
        if(command[i] == 'R'){
            dir++;
            dir %= 4;
        }
        else if(command[i] == 'L'){
            dir += 3;
            dir %= 4;
        }
        else if(command[i] == 'G'){
            answer[0] += move[dir][0];
            answer[1] += move[dir][1];
        }
        else if(command[i] == 'B'){
            answer[0] -= move[dir][0];
            answer[1] -= move[dir][1];
        }
    }
    return answer;
}
```

* **구현** 문제
* 나머지 연산으로 회전 처리, 좌회전 시 음수가 발생할 수 있으므로 `4`를 더해줌

### [[PCCP 모의고사 #2] 2번 - 신입사원 교육](https://school.programmers.co.kr/learn/courses/20848/lessons/255905)

```cpp
#include <string>
#include <vector>
#include <queue>
using namespace std;

int solution(vector<int> ability, int number) {
    int answer = 0;
    priority_queue<int, vector<int>, greater<int>> pq;
    for(int i=0; i<ability.size(); i++) pq.push(ability[i]);
    for(int i=0; i<number; i++){
        int a1 = pq.top(); pq.pop();
        int a2 = pq.top(); pq.pop();
        pq.push(a1 + a2); pq.push(a1 + a2); 
    }
    while(!pq.empty()){
        answer += pq.top();
        pq.pop();
    }
    return answer;
}
```

* **우선순위 큐**를 활용하는 **그리디** 문제 
* 항상 **최솟값 2개를 선택**하는 게 가장 이득
* 매번 순회하거나 정렬하면 시간복잡도가 각각 `O(n^2)`, `O(n^2logn)`이 되어버림
* 우선순위 큐를 사용하면 `O(nlogn)`으로 해결 가능

### [[PCCP 모의고사 #2] 3번 - 카페 확장](https://school.programmers.co.kr/learn/courses/20848/lessons/255906)

```cpp
#include <string>
#include <vector>

using namespace std;

int solution(vector<int> menu, vector<int> order, int k) {
    int answer = 0;
    vector<int> list(order.size());
    list[0] = menu[order[0]];
    
    for(int i=1; i<order.size(); i++){
        int start = max(k*i, list[i-1]);
        list[i] = start + menu[order[i]];
    }
    
    int inCur = 0, outCur = 0;
    int curr = 0;
    while(inCur < order.size()){
        int time = inCur * k;
        curr++;
        inCur++;
        
        while(outCur < list.size() && list[outCur] <= time){
            curr--;
            outCur++;
        }
        answer = max(answer, curr);
    }
    
    return answer;
}
```

* **투 포인터**, **시뮬레이션** 유형의 문제
* 각 손님이 나가는 시간을 미리 계산해 저장
* 이후 **투 포인터**로 다음 손님이 입장하는 시점에 나간 손님의 수를 감소

### [[PCCP 모의고사 #2] 4번 - 보물 지도](https://school.programmers.co.kr/learn/courses/20848/lessons/255907)

```cpp
#include <string>
#include <vector>
#include <queue>

using namespace std;

void BFS(const vector<vector<int>>& board, vector<vector<int>>& visit, pair<int, int> start){
    queue<pair<int, int>> q;
    int dy[4] = {-1, 1, 0, 0};
    int dx[4] = {0, 0, -1, 1};
    
    q.push(start);
    visit[start.first][start.second] = 0;
    
    while(!q.empty()){
        pair<int, int> curr = q.front();
        q.pop();
        int y = curr.first, x = curr.second;
        for(int i=0; i<4; i++){
            int ny = y + dy[i], nx = x + dx[i];
            if(ny<0 || ny>=board.size() || nx<0 || nx>=board[0].size() || 
               visit[ny][nx] != -1 || board[ny][nx] == 1) continue;
            visit[ny][nx] = visit[y][x] + 1;
            q.push({ny, nx});
        }
    }
}

int solution(int n, int m, vector<vector<int>> hole) {
    int answer = 1000*1000+1;
    vector<vector<int>> board(m, vector<int>(n, 0));
    vector<vector<int>> start(m, vector<int>(n, -1)), treasure(m, vector<int>(n, -1));
    for(int i=0; i<hole.size(); i++) 
        board[hole[i][1]-1][hole[i][0]-1] = 1;
    
    BFS(board, start, {0, 0});
    BFS(board, treasure, {m-1, n-1});
    
    if(start[m-1][n-1] != -1) answer = start[m-1][n-1] - 1;
    for(int i=0; i<hole.size(); i++){
        int y = hole[i][1] - 1, x = hole[i][0] - 1;
        if(y-1>=0 && y+1<m && board[y-1][x] == 0 && board[y+1][x] == 0){
            if(start[y-1][x] != -1 && treasure[y+1][x] != -1)
                answer = min(answer, start[y-1][x] + treasure[y+1][x] + 1);
            if(start[y+1][x] != -1 && treasure[y-1][x] != -1)
                answer = min(answer, start[y+1][x] + treasure[y-1][x] + 1);
        }
        
        if(x-1>=0 && x+1<n && board[y][x-1] == 0 && board[y][x+1] == 0){
            if(start[y][x-1] != -1 && treasure[y][x+1] != -1)
                answer = min(answer, start[y][x-1] + treasure[y][x+1] + 1);
            if(start[y][x+1] != -1 && treasure[y][x-1] != -1)
                answer = min(answer, start[y][x+1] + treasure[y][x-1] + 1);
        }
    }
        
    return answer == 1000*1000+1 ? -1 : answer;
}
```

* **BFS** 문제
* 시작점, 보물 위치에서 **각각 BFS를 실행**
* 이후 각 함정을 기준으로 함정을 점프로 넘어갈 수 있으면 거리 계산
* 이게 더 간단하다고 생각했는데 그냥 **상태 기반**으로 하는 게 더 빨랐을 듯

### [Path With Minimum Effort](https://leetcode.com/problems/path-with-minimum-effort/description/)

```cpp
#include <queue>
#include <cmath>

class Solution {
public:
    int dy[4] = {-1, 1, 0, 0};
    int dx[4] = {0, 0, -1, 1};

    bool BFS(const vector<vector<int>>& heights, int k){
        vector<vector<int>> visit(heights.size(), vector<int>(heights[0].size(), 0));
        queue<pair<int, int>> q;
        q.push({0, 0});
        visit[0][0] = 1;

        while(!q.empty()){
            pair<int, int> curr = q.front();
            q.pop();
            int y = curr.first, x = curr.second;
            int height = heights[y][x];
            for(int i=0; i<4; i++){
                int ny = y + dy[i], nx = x + dx[i];
                if(ny<0 || ny>=heights.size() || nx<0 || nx>=heights[0].size() || 
                visit[ny][nx] == 1 || abs(height - heights[ny][nx]) > k) continue;
                visit[ny][nx] = 1;
                q.push({ny, nx});
                if(ny == heights.size()-1 && nx == heights[0].size()-1) return true;
            }
        }
        return false;
    }

    int minimumEffortPath(vector<vector<int>>& heights) {
        int left = 0, right = 1e6;
        int answer = 0;
        while(left<=right){
            int mid = (right - left) / 2 + left;
            if(BFS(heights, mid)){
                answer = mid;
                right = mid - 1;
            }
            else left = mid + 1;
        }
        return answer;
    }
};
```

* **BFS**, **Parametric Search**로 풀이
* 문제 의도는 **다익스트라**로 푸는 것 같지만, 이게 구현이 더 빠르다고 판단
