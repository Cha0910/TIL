## 2. 프로그래머스 문제 풀이

### [전력망을 둘로 나누기](https://school.programmers.co.kr/learn/courses/30/lessons/86971)

```cpp
#include <string>
#include <vector>
#include <queue>

using namespace std;
int BFS(const vector<vector<int>>& graph, int a, int b){
    int count = 1;
    vector<int> visit(graph.size(), 0);
    queue<int> q;
    visit[0] = 1;
    q.push(0);
    
    while(!q.empty()){
        int curr = q.front();
        q.pop();
        for(int i=0; i<graph[curr].size(); i++){
            int next = graph[curr][i];
            if(visit[next] || (curr == a && next == b) || (curr == b && next == a)) continue;
            visit[next] = 1;
            q.push(next);
            count++;
        }
    }
    
    return count;
}

int solution(int n, vector<vector<int>> wires) {
    int answer = 1000;
    vector<vector<int>> graph(n);
    
    for(int i=0; i<wires.size(); i++){
        graph[--wires[i][0]].push_back(--wires[i][1]);
        graph[wires[i][1]].push_back(wires[i][0]);
    }
    
    for(int i=0; i<wires.size(); i++){
        int result = BFS(graph, wires[i][0], wires[i][1]);
        answer = min(answer, abs(2*result-n)); 
    }
    return answer;
}
```

* **BFS**, **완전 탐색** 문제
* 각 간선을 제거한 후 BFS를 실행해 연결 요소 개수 카운팅
