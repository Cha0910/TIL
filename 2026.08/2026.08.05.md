## 2. 프로그래머스 문제 풀이

### [합승 택시 요금](https://school.programmers.co.kr/learn/courses/30/lessons/72413)

```cpp
#include <string>
#include <vector>
#include <queue>
#include <climits>

using namespace std;

vector<int> dijkstra(const vector<vector<pair<int, int>>>& graph, int start){
    vector<int> dist(graph.size(), INT_MAX/2);
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    pq.push({0, start});
    dist[start] = 0;
    
    while(!pq.empty()){
        pair<int, int> curr = pq.top();
        pq.pop();
        int node = curr.second;
        int weight = curr.first;
        if(dist[node] < weight) continue;
        
        for(int i=0; i<graph[node].size(); i++){
            int next = graph[node][i].second;
            int nextWeight = graph[node][i].first + weight;
            if(dist[next] < nextWeight) continue;
            dist[next] = nextWeight;
            pq.push({nextWeight, next});
        }
    }
    
    return dist;
}

int solution(int n, int s, int a, int b, vector<vector<int>> fares) {
    int answer = INT_MAX / 2;
    vector<vector<pair<int, int>>> graph(n);
    
    for(int i=0; i<fares.size(); i++){
        int n1 = fares[i][0] - 1, n2 = fares[i][1] - 1;
        int weight = fares[i][2];
        graph[n1].push_back({weight, n2});
        graph[n2].push_back({weight, n1});
    }
    vector<int> distS = dijkstra(graph, s-1);
    vector<int> distA = dijkstra(graph, a-1);
    vector<int> distB = dijkstra(graph, b-1);
    
    for(int i=0; i<n; i++){
        if(distS[i] == INT_MAX/2 || distA[i] == INT_MAX/2 || distB[i] == INT_MAX/2) continue;
        answer = min(answer, distS[i] + distA[i] + distB[i]);
    }
    return answer;
}
```

* **다익스트라** 문제
* **택시를 내리는 지점**을 `m`이라 하자
* 최종적으로 구하는 값은 `s -> m`, `m -> a`, `m -> b` 의 비용의 합
* 문제의 그래프는 **양방향**이므로 구하는 비용은 `s -> m`, `a -> m`, `b -> m`의 합과 같음
* 따라서 `s`, `a`, `b` 각각의 지점에서 다익스트라를 실행한 후 합이 최소가 되는 `m`을 찾으면 정답
